- 데이터베이스/테이블 생성
    # 각 줄마다 코드의 커서를 두고 컨트롤 엔터
    # 1. 데이터베이스 생성
    create database java2day09;
    # 2. 데이터베이스 활성화/사용
    use java2day09;
    # 3. 테이블/표 생성
    create table members(
        name text
    );
    # 4. 테이블 레코드 확인
    select * from members;

- 테이블 조작하기 : DML
    1. 레코드/행 삽입/추가
        INSERT INTO 테이블명(필드명1, 필드명2) VALUES( 값1 , 값2 )
    2. 레코드/행 조회/검색
        SELECT * FROM 테이블명
    3. 레코드/행 수정
        UPDATE 테이블명 SET 수정필드 = 새로운값
        UPDATE 테이블명 SET 수정필드 = 새로운값 WHERE 조건필드 = 찾은값
        # 조건필드의 값이 해당 찾은값 과 동일한 레코드 조회/검색. =같다
    4. 레코드/행 삭제
        DELETE FROM 테이블명
        DELETE FROM 테이블명 WHERE 조건필드 = 찾은값


    Connection
        .prepareStatement('실행할 SQL질의어') : 기재준비
        .executeUpdate() :  INSERT, UPDATE , DELETE 에서ㅏ 주로 사용
        .executeQuery(); : select 에서 주로 사용
            .rs.next()      : 결과 레코드중에서 다음 레코드 이동 , 없이면 반환
            .rs.getString("name")   :  현재 레코드중에서 'name' 필드의 값 호출
            .rs.getString( 1 )      : 현재 레코드코드중에서 , 1번 레코드의 'name' 필드의 값 호출
